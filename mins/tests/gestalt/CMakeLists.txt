cmake_minimum_required(VERSION 3.16)
project(gestalt_mins LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc video calib3d features2d imgcodecs)
find_package(Threads REQUIRED)
get_filename_component(MINS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT MINS_SOURCE)
  message(FATAL_ERROR "Supply the verified isolated MINS_SOURCE from control.nodes.mins.build")
endif()
file(GLOB_RECURSE MINS_SOURCES "${MINS_SOURCE}/mins/src/*.cpp")
file(GLOB_RECURSE CORE_SOURCES "${MINS_SOURCE}/thirdparty/open_vins/ov_core/src/*.cpp")
list(FILTER MINS_SOURCES EXCLUDE REGEX "/(run_[^/]*|ROS[^/]*|Sim[2]?Visualizer)\\.cpp$")
list(FILTER MINS_SOURCES EXCLUDE REGEX "/io/")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/(test_[^/]*|dummy)\\.cpp$")
add_library(mins STATIC ${MINS_SOURCES} ${CORE_SOURCES})
target_compile_definitions(mins PUBLIC ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=0)
# Regression programs assert the reported innovation and rejection diagnostics.
target_compile_definitions(mins PRIVATE GESTALT_DIAGNOSTIC_HOOKS=1)
target_include_directories(mins PUBLIC "${MINS_SOURCE}/mins/src"
  "${MINS_SOURCE}/thirdparty/open_vins/ov_core/src" ${OpenCV_INCLUDE_DIRS})
target_include_directories(mins PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mins PUBLIC Eigen3::Eigen ${OpenCV_LIBS} Threads::Threads)
add_executable(whitening_check whitening.cc)
target_include_directories(whitening_check PRIVATE "${MINS_SOURCE}/mins/src/update/cam")
target_link_libraries(whitening_check PRIVATE Eigen3::Eigen)
enable_testing()
add_test(NAME camera_covariance_whitening COMMAND whitening_check)
if(MINS_SOURCE)
  set(CPI_CORE "${MINS_SOURCE}/thirdparty/open_vins/ov_core/src")
  add_executable(cpi_covariance_check cpi_covariance.cc "${CPI_CORE}/cpi/CpiV1.cpp")
  target_include_directories(cpi_covariance_check PRIVATE "${CPI_CORE}")
  target_link_libraries(cpi_covariance_check PRIVATE Eigen3::Eigen)
  add_test(NAME forward_and_reverse_preintegration_covariance COMMAND cpi_covariance_check)
endif()
if(MINS_SOURCE)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc video calib3d features2d imgcodecs)
  find_package(Threads REQUIRED)
  foreach(CHECK IN ITEMS imu_selection clone_selection cpi_velocity innovation linear_algebra wheel_resume observation_history)
    add_executable(${CHECK}_check ${CHECK}.cc)
    target_include_directories(${CHECK}_check PRIVATE
      "${MINS_SOURCE}/mins/src" "${MINS_SOURCE}/thirdparty/open_vins/ov_core/src"
      ${OpenCV_INCLUDE_DIRS})
    target_compile_definitions(${CHECK}_check PRIVATE ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=0)
    target_link_libraries(${CHECK}_check PRIVATE mins Eigen3::Eigen
      ${OpenCV_LIBS} Threads::Threads)
  endforeach()
  add_test(NAME factorized_updates_preserve_information COMMAND linear_algebra_check)
  add_test(NAME preserve_every_interior_imu_sample COMMAND imu_selection_check)
  add_test(NAME nearest_clone_does_not_read_output_argument COMMAND clone_selection_check)
  add_test(NAME wheel_resumes_after_invalid_interval COMMAND wheel_resume_check)
  add_test(NAME stationary_history_survives_filter_consumption COMMAND observation_history_check)
  foreach(MODE IN ITEMS forward reintegrate linear)
    add_test(NAME cpi_velocity_${MODE} COMMAND cpi_velocity_check ${MODE})
  endforeach()
  foreach(MODE IN ITEMS gate update propagation)
    add_test(NAME innovation_validity_${MODE} COMMAND innovation_check ${MODE})
  endforeach()
  option(MINS_PERFORMANCE_CHECKS "Check optimized feature history cleanup" ON)
  if(MINS_PERFORMANCE_CHECKS)
    add_executable(feature_history_check feature_history.cc)
    target_include_directories(feature_history_check PRIVATE
      "${MINS_SOURCE}/mins/src" "${MINS_SOURCE}/thirdparty/open_vins/ov_core/src"
      ${OpenCV_INCLUDE_DIRS})
    target_compile_definitions(feature_history_check PRIVATE ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=0)
    target_link_libraries(feature_history_check PRIVATE mins Eigen3::Eigen
      ${OpenCV_LIBS} Threads::Threads)
    add_test(NAME feature_history_bounded_after_clone_marginalization COMMAND feature_history_check)
  endif()
endif()
